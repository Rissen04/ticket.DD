from PIL import Image, ImageDraw, ImageFont def create_ticket_stamp( input_image_path, output_image_path, timestamp="18:31", ticket_code="S-4", english_word="Sample", background_image_path="stamp_background.png" ): base_img = Image.open(background_image_path).convert("RGBA") input_img = Image.open(input_image_path).convert("RGBA") input_img = input_img.resize(base_img.size) merged = Image.alpha_composite(input_img, base_img) draw = ImageDraw.Draw(merged) try: font_large = ImageFont.truetype("arial.ttf", 80) font_medium = ImageFont.truetype("arial.ttf", 40) font_small = ImageFont.truetype("arial.ttf", 30) except: font_large = ImageFont.load_default() font_medium = ImageFont.load_default() font_small = ImageFont.load_default() center_x, center_y = base_img.size[0]//2, base_img.size[1]//2 radius_outer = 180 radius_inner = 145 draw.ellipse( [(center_x - radius_outer, center_y - radius_outer), (center_x + radius_outer, center_y + radius_outer)], outline="red", width=8 ) draw.ellipse( [(center_x - radius_inner, center_y - radius_inner), (center_x + radius_inner, center_y + radius_inner)], outline="red", width=8 ) entered_text = "入場済" text_w, text_h = draw.textsize(entered_text, font=font_medium) draw.text((center_x - text_w/2, center_y - radius_outer + 20), entered_text, fill="red", font=font_medium) draw.text((center_x - text_w/2, center_y + radius_outer - text_h - 20), entered_text, fill="red", font=font_medium) time_w, time_h = draw.textsize(timestamp, font=font_large) draw.text((center_x - time_w/2, center_y - time_h/2), timestamp, fill="red", font=font_large) code_text = f"{english_word} {ticket_code}" code_w, code_h = draw.textsize(code_text, font=font_small) draw.text((base_img.size[0] - code_w - 20, base_img.size[1] - code_h - 20), code_text, fill="red", font=font_small) merged.save(output_image_path) merged.show() if __name__ == "__main__": # 実行例（画像ファイル名は自分の環境に合わせて変えてください） create_ticket_stamp( input_image_path="your_uploaded_picture.jpg", output_image_path="ticket_with_stamp.png", timestamp="20:15", ticket_code="S-4", english_word="Sample", background_image_path="stamp_background.png" )
